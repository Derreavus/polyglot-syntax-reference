Getting Started

# Install toolchain
curl --proto '=https' --tlsv1.2 -sSf https://sh.rustup.rs | sh

# Check versions
rustc --version
cargo --version

# Create & run a project
cargo new my_project
cd my_project
cargo run
cargo build --release
cargo test
cargo check          # fast type-check, no binary
cargo clippy         # extra lints
cargo fmt            # format
cargo add serde      # add dependency
cargo doc --open     # generate & open docs

# Single-file compile (rare)
rustc main.rs
Prefer Cargo for almost everything. The full language book lives at doc.rust-lang.org/book.

Data Types

Rust is statically typed; the compiler usually infers types. Integers default to i32, floats to f64. char is a full Unicode scalar value (4 bytes).

// Scalar types
let i: i32 = 42;              // signed 32-bit
let u: u8 = 255;              // unsigned 8-bit
let f: f64 = 3.14;            // 64-bit float
let b: bool = true;
let c: char = 'R';            // Unicode scalar

// Integer sizes: i8/u8, i16/u16, i32/u32, i64/u64, i128/u128, isize/usize
// Floating: f32, f64

// Type conversion
"42".parse::().unwrap(); // 42
42.to_string();               // "42"
42 as f64;                    // 42.0 (as is a cast)

// Compound
let tup: (i32, f64, char) = (1, 2.0, 'a');
let (x, y, z) = tup;          // destructure
let arr: [i32; 3] = [1, 2, 3];
let zeros = [0; 5];           // [0, 0, 0, 0, 0]

// Unit type
let unit: () = ();

Variables & Assignment

Variables are immutable by default. Add mut to allow reassignment. Shadowing lets you reuse a name (even with a new type).

let name = "Leo";             // &str, immutable
let mut age = 7;              // mutable
age += 1;                     // OK

const MAX_AGE: u32 = 120;     // must have explicit type, compile-time
static GREETING: &str = "Hi!"; // 'static lifetime

// Shadowing
let x = 5;
let x = x * 2;                // still immutable, new binding
let x = x.to_string();        // new type is fine

// Destructuring assignment
let (a, b) = (1, 2);
let Point { x, y } = p;

Strings

String is owned, growable, heap-allocated. &str is a borrowed string slice (often a view into a String or a literal). Prefer &str for function parameters when you do not need ownership.

let s1: &str = "Hello";                 // borrowed slice
let s2 = String::from("World");
let s3 = "Hi".to_string();              // also owned

// Operations
"me".to_string() + "ow!";               // "meow!"
"Meow!".repeat(3);
"Python".len();                         // 6 (bytes, not chars)
"a".to_uppercase();                     // "A"
" a ".trim();                           // "a"
"abc".replace("bc", "ha");              // "aha"
"a b".split(' ').collect::>();   // ["a", "b"]
["a", "b"].join("-");                   // "a-b"

// Indexing is byte-based — be careful with Unicode
let text = "Python";
&text[0..1];                            // "P"
&text[..3];                             // "Pyt"
&text[3..];                             // "hon"
text.chars().rev().collect::(); // "nohtyP"

// Formatting
let name = "Aubrey";
let age = 2;
format!("Hello, {name}!");              // captured variables
format!("{name} is {age}");
println!("{:?}", age);                  // Debug
println!("{:#?}", complex);             // pretty Debug

Control Flow

Rust has no truthiness — conditions must be bool. if is an expression.

let category = if age < 13 {
    "child"
} else if age < 20 {
    "teenager"
} else {
    "adult"
};

// loop runs forever until break; can return a value
let mut count = 0;
let result = loop {
    count += 1;
    if count == 3 { continue; }         // skip rest of body
    if count == 10 { break count * 2; } // result = 20
};

// while
while condition { ... }

// for — anything implementing IntoIterator
for i in 0..5 { ... }                   // 0,1,2,3,4
for i in 0..=5 { ... }                  // inclusive
for fruit in &fruits { ... }
for (i, fruit) in fruits.iter().enumerate() {
    println!("{i}: {fruit}");
}

// Labeled loops
'outer: for x in 0..5 {
    for y in 0..5 {
        if x * y > 6 { break 'outer; }
        println!("{x},{y}");
    }
}

Functions & Closures

fn greet() -> String {
    "Hello!".to_string()
}

fn greet_person(name: &str) -> String {
    format!("Hello, {name}!")
}

fn add(x: i32, y: i32) -> i32 {
    x + y                               // no semicolon = return value
}

// Multiple return values via tuple
fn min_max(nums: &[i32]) -> (i32, i32) {
    (*nums.iter().min().unwrap(), *nums.iter().max().unwrap())
}
let (lo, hi) = min_max(&[1, 5, 3]);

// Closures
let square = |x: i32| x * x;
square(5);                              // 25

let nums = vec![1, 2, 3, 4];
let sq: Vec = nums.iter().map(|x| x * x).collect();

// Returning closures (move captures ownership)
fn make_adder(n: i32) -> impl Fn(i32) -> i32 {
    move |x| x + n
}
let add5 = make_adder(5);
add5(10);                               // 15

// Fn / FnMut / FnOnce traits describe what a closure can do

Ownership & Borrowing

Every value has exactly one owner. When the owner goes out of scope the value is dropped. Assigning a non-Copy value moves it; the old variable becomes invalid.

let s1 = String::from("hello");
let s2 = s1;                            // move — s1 is now invalid
// println!("{s1}");                    // compile error
println!("{s2}");                       // OK

let n1 = 5;                             // i32 is Copy
let n2 = n1;                            // copy — both remain valid
println!("{n1} {n2}");

// Borrowing — shared (immutable)
fn print_len(s: &String) -> usize { s.len() }
let s = String::from("hello");
print_len(&s);                          // borrow only, no ownership taken
println!("{s}");                        // s is still valid

// Mutable borrow — exclusive
fn add_world(s: &mut String) {
    s.push_str(", world");
}
let mut s = String::from("hello");
add_world(&mut s);

// Rules: either one &mut OR any number of & at a time — no data races

// Cloning when you really need a deep copy
let s3 = s2.clone();
RAII is automatic: resources are acquired when the value is created and released when it is dropped (end of scope, or explicit drop(v)).

Lifetimes

// 'a says: the returned reference lives as long as the shorter of the two inputs
fn longest<'a>(x: &'a str, y: &'a str) -> &'a str {
    if x.len() > y.len() { x } else { y }
}

// Lifetime elision covers most common cases so you often omit annotations.
// Structs that hold references need explicit lifetimes.
struct Excerpt<'a> {
    part: &'a str,
}

impl<'a> Excerpt<'a> {
    fn level(&self) -> i32 { 3 }
    fn announce_and_return_part(&self, announcement: &str) -> &str {
        println!("Attention: {announcement}");
        self.part
    }
}

// 'static is the lifetime of the entire program (string literals have it)

Structs & Enums

struct Dog {
    name: String,
    age: u8,
}

impl Dog {
    fn new(name: &str, age: u8) -> Self {
        Self { name: name.to_string(), age }
    }
    fn bark(&self) -> String {
        format!("{} says Woof!", self.name)
    }
    fn have_birthday(&mut self) {
        self.age += 1;
    }
}

let my_dog = Dog::new("Frieda", 3);
my_dog.bark();

// Tuple struct
struct Point(i32, i32);
let p = Point(3, 4);
p.0; p.1;

// Unit-like struct
struct AlwaysEqual;

// Enum — one of several variants (can hold data)
enum Animal {
    Dog(String),
    Cat { name: String, lives: u8 },
}
let pet = Animal::Cat { name: "Leo".into(), lives: 9 };

// Option and Result are just enums
let some_value: Option = Some(5);
let no_value: Option = None;

Traits & Generics

trait Shape {
    fn area(&self) -> f64;
    fn name(&self) -> &str { "shape" }  // default implementation
}

struct Circle { radius: f64 }
impl Shape for Circle {
    fn area(&self) -> f64 {
        std::f64::consts::PI * self.radius.powi(2)
    }
}

// Generic function with trait bounds
fn largest(list: &[T]) -> T {
    let mut max = list[0];
    for &item in list {
        if item > max { max = item; }
    }
    max
}
largest(&[3, 7, 2]);                    // 7

// impl Trait sugar (argument position)
fn print_shape(shape: &impl Shape) {
    println!("Area: {}", shape.area());
}

// Trait objects for dynamic dispatch / mixed types
let shapes: Vec> = vec![
    Box::new(Circle { radius: 1.0 }),
];
for s in &shapes { println!("{}", s.area()); }

// Derive common traits
#[derive(Debug, Clone, PartialEq, Eq, Hash, Default)]
struct Point { x: i32, y: i32 }

// Associated types, supertraits, etc. are also available

Pattern Matching

match n {
    1 => println!("one"),
    2 | 3 => println!("two or three"),
    4..=9 => println!("four through nine"),
    _ => println!("something else"),     // required for exhaustiveness
}

// Destructuring
let point = (0, 5);
match point {
    (0, 0) => println!("origin"),
    (0, y) => println!("on y-axis at {y}"),
    (x, _) => println!("x = {x}"),
}

// if let / while let — concise single-pattern
let maybe_value: Option = Some(7);
if let Some(v) = maybe_value {
    println!("Got {v}");
}

// Match guards
match n {
    x if x < 0 => println!("negative"),
    x if x % 2 == 0 => println!("even"),
    _ => println!("odd"),
}

// Struct destructuring & @ bindings
struct Point { x: i32, y: i32 }
let p = Point { x: 0, y: 7 };
let Point { x, y } = p;
match y {
    n @ 1..=9 => println!("single digit: {n}"),
    _ => println!("other"),
}

Collections

// Vec — growable array (most common choice)
let mut nums: Vec = Vec::new();
let mut nums = vec![5];                 // macro shorthand
nums.push(10);
nums.insert(0, 99);
let last = nums.pop();                  // Option
nums.remove(0);

let fruits = vec!["banana", "apple", "orange"];
fruits[0];                              // "banana"
fruits.contains(&"apple");              // true
fruits.len();

// Iterating & slicing
for fruit in &fruits { println!("{fruit}"); }
let mid = &fruits[1..];                 // ["apple", "orange"]

// HashMap / HashSet
use std::collections::{HashMap, HashSet};
let mut pet = HashMap::new();
pet.insert("name", "Leo");
pet.insert("age", "42");
pet.get("age");                         // Option<&V>
pet.insert("age", "7");                 // update
pet.remove("age");
*pet.entry("hits").or_insert(0) += 1;   // entry API

// BTreeMap / BTreeSet keep keys sorted
use std::collections::BTreeMap;
let mut scores = BTreeMap::new();
scores.insert("b", 2);
scores.insert("a", 1);
for (k, v) in &scores {                 // always a then b
    println!("{k}: {v}");
}

Iterators

Iterators are lazy — nothing happens until a consuming adapter is called.

let squares: Vec = (0..10).map(|x| x * x).collect();
let evens: Vec = (0..20).filter(|x| x % 2 == 0).collect();

let matrix: Vec> = (0..3)
    .map(|i| (0..3).map(|j| i * j).collect())
    .collect();

// Common adapters
let words = vec!["hello", "world"];
let lens: Vec = words.iter().map(|w| w.len()).collect();
let total: i32 = (1..=1000).sum();
words.iter().any(|w| w.len() > 4);      // true
words.iter().all(|w| w.len() < 10);     // true

// Collect into other types
use std::collections::HashMap;
let map: HashMap<&str, usize> = words.iter()
    .map(|w| (*w, w.len())).collect();

// fold, zip, chain
let product = (1..=5).fold(1, |acc, x| acc * x); // 120
let names = vec!["Leo", "Max"];
let ages = vec![7, 3];
let pairs: Vec<_> = names.iter().zip(ages.iter()).collect();
let all: Vec = (1..3).chain(7..9).collect(); // [1,2,7,8]

Error Handling

No exceptions. Recoverable errors use Result. Unrecoverable errors use panic!.

fn divide(a: f64, b: f64) -> Result {
    if b == 0.0 {
        return Err("Cannot divide by zero!".into());
    }
    Ok(a / b)
}

match divide(10.0, 0.0) {
    Ok(r) => println!("Result: {r}"),
    Err(e) => println!("{e}"),
}

// ? operator propagates Err (or None for Option)
fn read_number() -> Result> {
    let s = std::fs::read_to_string("num.txt")?;
    let n: i32 = s.trim().parse()?;
    Ok(n)
}

// Common shortcuts
result.unwrap();                        // panics if Err / None
result.unwrap_or(0);                    // fallback value
result.expect("custom message");

// Combinators instead of match
let doubled = Some(4).map(|n| n * 2);   // Some(8)
"5".parse::().map(|n| n + 1).unwrap_or(0); // 6

// Custom error types often use thiserror or anyhow crates

Concurrency

use std::thread;
use std::sync::{Arc, Mutex};
use std::sync::mpsc;

// Threads
let handle = thread::spawn(|| {
    println!("from another thread");
});
handle.join().unwrap();

// Channels
let (tx, rx) = mpsc::channel();
thread::spawn(move || {
    tx.send("hello").unwrap();
});
println!("{}", rx.recv().unwrap());

// Shared state with Arc>
let counter = Arc::new(Mutex::new(0));
let mut handles = vec![];
for _ in 0..10 {
    let c = Arc::clone(&counter);
    handles.push(thread::spawn(move || {
        let mut num = c.lock().unwrap();
        *num += 1;
    }));
}
for h in handles { h.join().unwrap(); }

// async/await (usually with tokio)
// #[tokio::main]
// async fn main() { ... }

Advanced Idioms

// Newtype pattern
struct Meters(f64);
struct Seconds(f64);

// Type aliases
type Result = std::result::Result>;

// PhantomData for unused type parameters
use std::marker::PhantomData;
struct Parser { _marker: PhantomData }

// Interior mutability
use std::cell::RefCell;
let data = RefCell::new(5);
*data.borrow_mut() += 1;

// Cow (clone-on-write)
use std::borrow::Cow;
fn process(input: Cow) { ... }
