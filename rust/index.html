Gotchas

// Unnecessary clone — prefer a borrow
let s2 = s1.clone();

// Borrowing across mutation
let r = &v[0];
v.push(1);             // error while r lives

// unwrap in libraries — prefer ? / unwrap_or
value.unwrap();

// Blocking inside async — use tokio::time::sleep
std::thread::sleep(...);

// Prefer &T / &mut T; Arc only for shared ownership across threads
Common mistake clone() and unwrap() silence the compiler but hide design issues.
