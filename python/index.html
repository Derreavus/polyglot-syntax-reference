Gotchas

# Mutable default — shared across calls
def append(item, bucket=[]):   # BAD
    bucket.append(item); return bucket

def append(item, bucket=None):  # GOOD
    if bucket is None: bucket = []
    bucket.append(item); return bucket

# is vs ==
a is b          # identity
a == b          # equality

# Late-binding closures
funcs = [lambda: i for i in range(3)]
funcs[0]()      # 2 — all see final i
funcs = [lambda i=i: i for i in range(3)]  # capture

# Generator exhaustion
g = (x*x for x in range(3))
list(g); list(g)  # second is []

# Prefer specific exceptions over bare except
Common mistake Mutable defaults and late-binding closures are classic traps. Prefer None defaults and default-arg capture in lambdas.
