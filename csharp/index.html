LINQ

    .Where(p => p.Age >= 18)
    .OrderBy(p => p.LastName)
    .Select(p => p.Name)

        where p.Age >= 18

Async / Await

public async Task FetchAsync(string url) {
    using var client = new HttpClient();
    return await client.GetStringAsync(url);
}

public async Task ProcessManyAsync(IEnumerable urls) {
    var tasks = urls.Select(FetchAsync);
    string[] results = await Task.WhenAll(tasks);
}

// ValueTask for hot paths that often complete synchronously
public async ValueTask GetCachedAsync() { ... }

// IAsyncEnumerable (C# 8+)
public async IAsyncEnumerable GenerateAsync() {
    for (int i = 0; i < 10; i++) {
        await Task.Delay(100);
        yield return i;
    }
}

await foreach (var n in GenerateAsync()) { ... }

Pattern Matching

// Type pattern + property pattern
if (obj is Person { Age: > 18, Name: var n }) {
    Console.WriteLine(n);
}

// Switch expression with patterns
string Describe(object o) => o switch {
    int i when i > 0 => "positive int",
    string { Length: > 0 } s => $"non-empty {s}",
    Person { Age: >= 18 } => "adult",
    null => "null",
    _ => "something else"
};

// List patterns (C# 11)
if (arr is [1, 2, .. var rest]) { ... }
if (arr is [var first, .., var last]) { ... }

// Relational, logical, parenthesized patterns also supported
