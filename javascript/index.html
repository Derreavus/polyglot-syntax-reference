Data Types

JavaScript has seven primitive types. Objects, arrays, and functions are reference values.

const count = 42;
const ratio = 3.14;
const text = "hello";
const enabled = true;
const missing = undefined;
const empty = null;
const symbol = Symbol("id");
const large = 123n;

Variables & Assignment

Use const by default, let when reassignment is needed, and avoid var in new code.

const name = "Ada";
let count = 0;
count += 1;
const { name: userName } = user;
const [first, ...rest] = values;

Operators

+ - * / % **
= += -= *= /=
=== !== == != < > <= >=
&& || ! ?? ?.
& | ^ ~ << >>
value ?? fallback;
user?.address?.city;

Prefer === and !== to avoid coercive equality.

Control Flow

if (score >= 90) {
  grade = "A";
} else if (score >= 80) {
  grade = "B";
} else {
  grade = "C";
}

for (const item of items) {
  console.log(item);
}

switch (status) {
  case "ready": start(); break;
  default: wait();
}

Functions

function add(a, b = 0) {
  return a + b;
}

const square = (n) => n * n;
const logAll = (...values) => console.log(...values);

function makeCounter() {
  let count = 0;
  return () => ++count;
}

Objects & Classes

const point = { x: 2, y: 3 };
point.x;

class Animal {
  constructor(name) { this.name = name; }
  speak() { return `${this.name} makes a sound`; }
}

class Dog extends Animal {
  speak() { return `${this.name} barks`; }
}

Collections

Async & Await

async function loadUser(id) {
  const response = await fetch(`/users/${id}`);
  if (!response.ok) throw new Error("Request failed");
  return response.json();
}

const [user, settings] = await Promise.all([
  loadUser("u1"),
  loadSettings(),
]);
